def krippendorff_alpha_ordinal(ratings_matrix):
    """
    Compute Krippendorff's alpha for ordinal data.
    ratings_matrix: DataFrame with items as rows, raters as columns.
    Uses squared ordinal distance metric.
    """
    values = sorted(ratings_matrix.stack().unique())
    val_map = {v: i for i, v in enumerate(values)}

    unit_counts = []
    for _, row in ratings_matrix.iterrows():
        vals = row.dropna().values
        if len(vals) < 2:
            continue
        counts = {}
        for v in vals:
            counts[v] = counts.get(v, 0) + 1
        unit_counts.append((len(vals), counts))

    if not unit_counts:
        return np.nan

    total_disagreement = 0
    total_pairs = 0
    marginals = {}

    for m_u, counts in unit_counts:
        for c, n_c in counts.items():
            marginals[c] = marginals.get(c, 0) + n_c
            for k, n_k in counts.items():
                if c != k:
                    dist = (val_map[c] - val_map[k]) ** 2
                    total_disagreement += n_c * n_k * dist
            total_pairs += n_c * (m_u - 1)

    if total_pairs == 0:
        return np.nan

    Do = total_disagreement / total_pairs

    n_total = sum(marginals.values())
    De = 0
    for c in marginals:
        for k in marginals:
            if c != k:
                dist = (val_map[c] - val_map[k]) ** 2
                De += marginals[c] * marginals[k] * dist
    De /= (n_total * (n_total - 1))

    if De == 0:
        return 1.0
    return 1 - Do / De

def pairwise_agreement(df, item_col, rater_col, label_col,
                       agg_func='mode', metric='accuracy'):
    """
    Compute pairwise inter-rater agreement matrix.

    Parameters
    ----------
    df : DataFrame
    item_col, rater_col, label_col : str
    agg_func : 'mode' or 'mean' — how to aggregate across seeds/temps
    metric : 'accuracy' or 'kappa'

    Returns
    -------
    pd.DataFrame — symmetric matrix of pairwise agreement
    """
    import pandas as pd
    from sklearn.metrics import cohen_kappa_score

    if agg_func == 'mode':
        agg = df.groupby([item_col, rater_col])[label_col].agg(
            lambda x: x.mode().iloc[0] if len(x.mode()) > 0 else x.iloc[0]
        ).reset_index()
    else:
        agg = df.groupby([item_col, rater_col])[label_col].mean().reset_index()

    pivot = agg.pivot(index=item_col, columns=rater_col, values=label_col)
    raters = sorted(pivot.columns)

    mat = pd.DataFrame(np.nan, index=raters, columns=raters)
    for i, r1 in enumerate(raters):
        mat.loc[r1, r1] = 1.0
        for r2 in raters[i + 1:]:
            valid = pivot[[r1, r2]].dropna()
            if len(valid) == 0:
                continue
            if metric == 'kappa':
                try:
                    k = cohen_kappa_score(valid[r1], valid[r2], weights='quadratic')
                except:
                    k = np.nan
                mat.loc[r1, r2] = round(k, 4)
                mat.loc[r2, r1] = round(k, 4)
            else:
                acc = (valid[r1] == valid[r2]).mean()
                mat.loc[r1, r2] = round(acc, 4)
                mat.loc[r2, r1] = round(acc, 4)
    return mat

def calibration_metrics(df, prob_cols, label_col, labels, n_bins=10):
    """
    Compute Expected Calibration Error (ECE) and per-class calibration data.

    Returns dict with 'ece' and 'per_class' calibration curves.
    """
    results = {'ece': 0, 'per_class': {}}
    total_samples = len(df)

    for label, prob_col in zip(labels, prob_cols):
        is_positive = (df[label_col] == label).astype(float).values
        probs = df[prob_col].values.astype(float)

        # Bin probabilities
        bin_edges = np.linspace(0, 1, n_bins + 1)
        bin_centers = []
        bin_accs = []
        bin_confs = []
        bin_counts = []

        for b in range(n_bins):
            mask = (probs >= bin_edges[b]) & (probs < bin_edges[b + 1])
            if b == n_bins - 1:
                mask = (probs >= bin_edges[b]) & (probs <= bin_edges[b + 1])
            if mask.sum() == 0:
                continue
            bin_acc = is_positive[mask].mean()
            bin_conf = probs[mask].mean()
            bin_centers.append((bin_edges[b] + bin_edges[b + 1]) / 2)
            bin_accs.append(bin_acc)
            bin_confs.append(bin_conf)
            bin_counts.append(mask.sum())

        results['per_class'][label] = {
            'bin_centers': bin_centers,
            'bin_accs': bin_accs,
            'bin_confs': bin_confs,
            'bin_counts': bin_counts,
        }

    # Overall ECE: weighted average |acc - conf| across all bins and classes
    total_ece = 0
    total_n = 0
    for label_data in results['per_class'].values():
        for acc, conf, cnt in zip(label_data['bin_accs'], label_data['bin_confs'],
                                   label_data['bin_counts']):
            total_ece += cnt * abs(acc - conf)
            total_n += cnt
    results['ece'] = total_ece / max(total_n, 1)

    return results

Three entropy–θ correlations test whether model-generated uncertainty predicts item difficulty:

## Per-Item Difficulty (θ Proxy) & Entropy–θ Correlations

The mean severity per item, averaged over all 216 observations (4 evaluators × 3 prompts × 6 temps × 3 seeds), serves as a sufficient statistic for the Rasch item difficulty parameter θ.

Three entropy–θ correlations test whether model-generated uncertainty predicts item difficulty:
1. **Baseline entropy ↔ mean severity** — baseline = gpt-4o-mini × rubric × temp=0 (gpt-4o-mini screening condition)
2. **All-condition entropy ↔ mean severity** — entropy averaged over all 216 observations
3. **Baseline entropy ↔ all-condition entropy** — validates single-call entropy as a proxy for multi-evaluator entropy

# ── Per-Item Difficulty (θ proxy) & Entropy–θ Correlations ──
from scipy.stats import spearmanr, pearsonr

# Probability columns for Shannon entropy (7-class)
_prob_cols = ['prob_normal', 'prob_depression', 'prob_anxiety', 'prob_suicidal',
              'prob_stress', 'prob_bipolar', 'prob_personality_disorder']

def _shannon_h(row, cols):
    probs = [row[c] for c in cols if pd.notna(row[c]) and row[c] > 0]
    return -sum(p * np.log2(p) for p in probs) if probs else 0.0

# 1. Per-item mean severity (θ proxy): averaged over all 216 observations
item_theta = results.groupby('item_id')['severity_7'].mean().reset_index()
item_theta.columns = ['item_id', 'mean_severity']

# 2. Baseline entropy: gpt-4o-mini × rubric × temp=0 (gpt-4o-mini screening condition)
ref_single = results[
    (results['evaluator'] == 'gpt-4o-mini') &
    (results['prompt_type'] == 'rubric') &
    (results['temperature'].astype(float) == 0.0)
].copy()
ref_single['_h'] = ref_single.apply(lambda r: _shannon_h(r, _prob_cols), axis=1)
baseline_h = ref_single.groupby('item_id')['_h'].mean().reset_index()
baseline_h.columns = ['item_id', 'baseline_entropy']

# 3. All-condition entropy: averaged over all 216 observations per item
results['_h_all'] = results.apply(lambda r: _shannon_h(r, _prob_cols), axis=1)
allcond_h = results.groupby('item_id')['_h_all'].mean().reset_index()
allcond_h.columns = ['item_id', 'allcond_entropy']

# Merge into item-level DataFrame
item_df = item_theta.merge(baseline_h, on='item_id').merge(allcond_h, on='item_id')
_samp = sample[['item_id', 'entropy_stratum']].copy()
_samp['item_id'] = _samp['item_id'].astype(str)
item_df['item_id'] = item_df['item_id'].astype(str)
item_df = item_df.merge(_samp, on='item_id', how='left')

print(f"Items: {len(item_df)}")
print(f"\n{'='*70}")
print("Per-Item θ Proxy (Mean Severity) & Entropy Summary")
print(f"{'='*70}")
print(item_df[['mean_severity', 'baseline_entropy', 'allcond_entropy']].describe().round(3).to_string())

# ── Three Correlations (Spearman + Pearson) ──
print(f"\n{'='*70}")
print("Entropy–θ Correlations")
print(f"{'='*70}")
corr_specs = [
    ('Baseline Entropy ↔ Mean Severity', 'baseline_entropy', 'mean_severity'),
    ('All-Condition Entropy ↔ Mean Severity', 'allcond_entropy', 'mean_severity'),
    ('Baseline Entropy ↔ All-Condition Entropy', 'baseline_entropy', 'allcond_entropy'),
]
corr_rows = []
for label, c1, c2 in corr_specs:
    rho, p_rho = spearmanr(item_df[c1], item_df[c2])
    r, p_r = pearsonr(item_df[c1], item_df[c2])
    corr_rows.append({'Comparison': label,
                      'Spearman ρ': round(rho, 3), 'p (Spearman)': f'{p_rho:.2e}',
                      'Pearson r': round(r, 3), 'p (Pearson)': f'{p_r:.2e}'})
    print(f"  {label}:")
    print(f"    Spearman ρ = {rho:.3f} (p = {p_rho:.2e})")
    print(f"    Pearson  r = {r:.3f} (p = {p_r:.2e})")
corr_table_theta = pd.DataFrame(corr_rows)
display(corr_table_theta)

fig, ax = plt.subplots(figsize=(8, 6))
