Mental Health: descriptive cross-trait diagnostics

Retained item-centered and evaluator-half comparisons; binary-scale checks for the six original flags.

Scope and saved-analysis status

These are descriptive calculations on the original annotation arrays. Their saved values are preserved from the earlier reports. An isolated validation executed the focused source against the same annotation CSVs and reproduced the displayed diagnostics without replacing the saved outputs. They are separate from both the primary joint OpenMx G-study and the new supplementary stacked lme4 G-study. No attenuation-adjusted correlations, old composite estimates, or historical G-theory bootstrap results are presented here.

Within-item correlations subtract each trait’s item mean and therefore combine several instrumentation and residual sources. Same-half and split-half comparisons average all three balanced partitions of four evaluators and both trait assignments. They change evaluator composition and call pairing while retaining prompts; their difference is not an identified proportion of shared-method error.

Binary-scale checks

These checks cover the six original flags. An item is positive when at least 108 of its 216 annotations flag the condition. Fréchet bounds concern manifest binary correlations. Probit correlations transform bounded item endorsement proportions; tetrachoric correlations use the item-level 2 × 2 table. A tetrachoric estimate is flagged when either condition occurs in fewer than ten items. The new three-group outcome is not part of these saved diagnostics.

```{r load-data}
#' One trait in long form under standard facet names.
# The CSV is read once and cached: build_trait() is called once per trait, so a
# plain read.csv() here re-parses the same 21,600-row file four or six times.
.trait_cache <- new.env(parent = emptyenv())
read_once <- function(path) {
  key <- normalizePath(path, mustWork = TRUE)
  if (is.null(.trait_cache[[key]])) .trait_cache[[key]] <- read.csv(key, stringsAsFactors = FALSE)
  .trait_cache[[key]]
}

build_trait <- function(data_dir, col) {
  d <- read_once(file.path(data_dir, "mh_labeling_final.csv"))
  x <- tolower(trimws(as.character(d[[col]])))
  v <- ifelse(x %in% c("true", "1"), 1,
       ifelse(x %in% c("false", "0"), 0, NA_real_))
  if (any(is.na(v))) stop("unmapped values in ", col)
  data.frame(score = v, item = factor(d$item_id),
             evaluator = factor(d$evaluator), prompt = factor(d$prompt_type),
             temp = factor(d$temperature), seed = factor(d$seed),
             stringsAsFactors = FALSE)
}

TD <- lapply(TRAITS, function(col) build_trait(DATA_DIR, col))
```

```{r validate-array}
#' Reshape a complete balanced long frame into a five-way array, facets on axes
#' in FACETS order (item first).  Errors rather than silently filling.
to_array <- function(df, score_col = "score") {
  levs <- lapply(FACETS, function(f) sort(unique(df[[f]])))
  names(levs) <- FACETS
  dims <- vapply(levs, length, integer(1))
  idx  <- vapply(FACETS, function(f) match(df[[f]], levs[[f]]),
                 integer(nrow(df)))
  lin  <- idx[, 1]
  mult <- 1
  for (j in seq_along(FACETS)[-1]) {
    mult <- mult * dims[j - 1]
    lin  <- lin + (idx[, j] - 1) * mult
  }
  if (anyDuplicated(lin))        stop("duplicate factor combinations")
  if (length(lin) != prod(dims)) stop("design is not complete/balanced")
  arr <- array(NA_real_, dim = unname(dims))
  arr[lin] <- df[[score_col]]
  if (anyNA(arr))                stop("design is not complete/balanced")
  arr
}
```

```{r descriptive-cross-trait}
# Item-mean correlations are calculated directly from the annotations here.
# The historical source used proportional-shrinkage BLUP correlations and
# asserted their equality to these item-mean correlations to tolerance 1e-8.
# The trait frames come from one CSV in one row order; the split below indexes
# every trait with a mask built on the first, so that must hold.
ALIGN_FACETS <- c("item", "evaluator", "prompt", "temp", "seed")
stopifnot(all(vapply(TD, function(x)
  identical(x[ALIGN_FACETS], TD[[1]][ALIGN_FACETS]), logical(1))))
TRAIT_PAIRS <- combn(names(TRAITS), 2, simplify = FALSE)

resid_ic <- sapply(names(TRAITS), function(tn) {
  x <- TD[[tn]]; x$score - ave(x$score, x$item, FUN = mean)
})
R_err   <- cor(resid_ic)

evs <- sort(unique(as.character(TD[[1]]$evaluator)))
stopifnot(length(evs) == 4L)
item_mean <- function(keep) sapply(names(TRAITS), function(tn) {
  x <- TD[[tn]]; tapply(x$score[keep], x$item[keep], mean)
})
all_m  <- item_mean(rep(TRUE, nrow(TD[[1]])))
# The three distinct 2-vs-2 splits (the other three combinations are their complements).
splits <- combn(evs, 2, simplify = FALSE)[1:3]
halves <- lapply(splits, function(A) {
  inA <- TD[[1]]$evaluator %in% A
  list(A = item_mean(inA), B = item_mean(!inA))
})
mv_samecall_vs_splitcall <- do.call(rbind, lapply(TRAIT_PAIRS, function(pr) {
  a <- pr[1]; b <- pr[2]
  cross  <- unlist(lapply(halves, function(h) c(cor(h$A[, a], h$B[, b]), cor(h$B[, a], h$A[, b]))))
  within <- unlist(lapply(halves, function(h) c(cor(h$A[, a], h$A[, b]), cor(h$B[, a], h$B[, b]))))
  data.frame(dataset = DATASET, trait_1 = a, trait_2 = b,
             r_same_call_all  = cor(all_m[, a], all_m[, b]),
             r_same_call_half = mean(within),
             r_split_call     = mean(cross),
             r_split_call_sd  = sd(cross),
             shared_method_inflation = mean(within) - mean(cross), row.names = NULL)
}))


mv_cross_trait_error_corr <- do.call(rbind, lapply(TRAIT_PAIRS, function(pr) {
  data.frame(dataset = DATASET, trait_1 = pr[1], trait_2 = pr[2],
             cross_trait_error_r = R_err[pr[1], pr[2]],
             r_item_mean = cor(all_m[, pr[1]], all_m[, pr[2]]), row.names = NULL)
```

```{r binary-scale-checks}
#' Fréchet bounds on the phi correlation of two binaries with marginals p, q.
phi_bounds <- function(p, q) {
  if (min(p, q) <= 0 || max(p, q) >= 1) return(c(lo = NA_real_, hi = NA_real_))
  den <- sqrt(p * (1 - p) * q * (1 - q))
  c(lo = (max(0, p + q - 1) - p * q) / den, hi = (min(p, q) - p * q) / den)
}
#' Maximum-likelihood tetrachoric correlation of two binary vectors.
tetrachoric <- function(x, y) {
  n11 <- sum(x == 1 & y == 1); n10 <- sum(x == 1 & y == 0)
  n01 <- sum(x == 0 & y == 1); n00 <- sum(x == 0 & y == 0)
  n <- n11 + n10 + n01 + n00
  px <- (n11 + n10) / n; py <- (n11 + n01) / n
  if (min(px, py) <= 0 || max(px, py) >= 1) return(NA_real_)
  hx <- qnorm(px); hy <- qnorm(py)
  negll <- function(r) {
    r <- max(min(r, 0.999), -0.999)
    p11 <- pbivnorm(hx, hy, rho = r)
    p11 <- max(min(p11, min(px, py) - 1e-12), 1e-12)
    ps  <- pmax(c(p11, px - p11, py - p11, 1 - px - py + p11), 1e-12)
    -(n11 * log(ps[1]) + n10 * log(ps[2]) + n01 * log(ps[3]) + n00 * log(ps[4]))
  }
  optimize(negll, interval = c(-0.99, 0.99))$minimum
}

prop   <- as.data.frame(sapply(names(TRAITS), function(tn) tapply(TD[[tn]]$score, TD[[tn]]$item, mean)))
n_obs  <- nrow(TD[[1]]) / nlevels(TD[[1]]$item)
tie_count <- vapply(prop, function(v) sum(v == 0.5), integer(1))
binmaj <- as.data.frame(lapply(prop, function(v) as.integer(v >= 0.5)))
probit <- as.data.frame(lapply(prop, function(v) qnorm(pmin(pmax(v, 0.5 / n_obs), 1 - 0.5 / n_obs))))
cat("Exact 50/50 ties assigned present:",
    paste(names(tie_count), tie_count, sep = "=", collapse = ", "), "\n")

mv_tetrachoric <- do.call(rbind, lapply(TRAIT_PAIRS, function(pr) {
  a <- pr[1]; b <- pr[2]
  pa <- mean(binmaj[[a]]); pb <- mean(binmaj[[b]]); bnd <- phi_bounds(pa, pb)
  n_pos <- min(sum(binmaj[[a]]), sum(binmaj[[b]]))
  phi <- if (length(unique(binmaj[[a]])) > 1 && length(unique(binmaj[[b]])) > 1)
    cor(binmaj[[a]], binmaj[[b]]) else NA_real_
  data.frame(dataset = DATASET, trait_1 = a, trait_2 = b,
             base_rate_1 = pa, base_rate_2 = pb, min_positive_items = n_pos,
             pearson_phi = phi, phi_min = bnd[["lo"]], phi_max = bnd[["hi"]],
             position_in_range = (phi - bnd[["lo"]]) / (bnd[["hi"]] - bnd[["lo"]]),
             at_lower_bound = isTRUE(abs(phi - bnd[["lo"]]) < 0.005),
             n_cooccur = sum(binmaj[[a]] == 1 & binmaj[[b]] == 1),
             tetrachoric = tetrachoric(binmaj[[a]], binmaj[[b]]),
             probit_scale_r = cor(probit[[a]], probit[[b]]),
             tetrachoric_unstable = n_pos < 10, row.names = NULL)
}))
knitr::kable(mv_tetrachoric, digits = 4, row.names = FALSE,
             caption = "Discretization ceiling: observed phi against its attainable range, with tetrachoric and probit-scale alternatives.")
```
